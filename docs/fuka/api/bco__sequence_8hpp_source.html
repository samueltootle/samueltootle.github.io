236  std::string const branch_name{"binary"};

241 
242  auto const& node_map{MBCO};
243 
244  std::array<NODES, 2> bcos{NODES::BCO1, NODES::BCO2};
245  for (const auto& node : branch) {

249

251  continue;

253  // remove suffix (e,g, bh1 -> bh, ns1 -> ns)
254  auto const tstr = node_str.substr(0, 2);
255 
256  const auto& it = node_map.find(tstr);
257 
258  auto const default_mass_param = (tstr == "bh") ? "mch" : "madm";
259  auto const default_spin_param = "chi";
260 
261  Tree const child_branch = read_branch(branch, node_str);
262 
263  if (it != node_map.end()) {
264  auto res = find_fixing_values(child_branch,
265  MBCO_PARAMS,
266  default_mass_param,
267  default_spin_param,
268  bco);
269  return res;
270  }
271  }
