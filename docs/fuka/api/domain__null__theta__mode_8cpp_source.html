24 void Domain::zero_top_theta_coef (Tensor& tt, int dom, int mquant) const {
25 
26  for (int c=0 ; c<tt.get_n_comp() ; c++) {
27  Val_domain& so = (tt.get_valence()==0) ? tt.set().set_domain(dom)
28  : tt.set(tt.indices(c)).set_domain(dom) ;
29  if (so.check_if_zero())
30  continue ;
31  so.coef() ;
32  int nt = nbr_coefs(1) ;
33  int baset = (*so.get_base().get_base_1d(1)) (0) ;
34  if (!is_null_theta_mode (baset, mquant, nt-1, nt))
35  continue ;
36 
37  Index pos (nbr_coefs) ;
38  do {
39  if ((pos(1)==nt-1) && ((ndim<3) || (pos(2)==0)))
40  so.set_coef(pos) = 0. ;
41  }
42  while (pos.inc()) ;
43  }
44 }
