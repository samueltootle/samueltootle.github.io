112  auto check_and_update_config = [&](auto p) {
113  exists = true;
114  base_config_t old_solution(p + ".info");
115 
116  if (bconfig->set(BCO_PARAMS::NSHELLS, NODES::BCO1) !=
117  old_solution.set(BCO_PARAMS::NSHELLS, NODES::BCO1) ||
118  bconfig->set(BCO_PARAMS::NSHELLS, NODES::BCO2) !=
119  old_solution.set(BCO_PARAMS::NSHELLS, NODES::BCO2))
120  return false;
121 
122  // make sure we copy stages, controls, and settings over
123  for (auto idx = 0; idx < STAGES::NUM_STAGES; ++idx)
124  old_solution.set_stage(idx) = bconfig->set_stage(idx);
125  for (auto idx : CONTROLS_ARY)
126  old_solution.control(idx) = bconfig->control(idx);
127  for (auto idx = 0; idx < SEQ_SETTINGS::NUM_SEQ_SETTINGS; ++idx)
128  old_solution.seq_setting(idx) = bconfig->seq_setting(idx);
129 
130  auto& stages = bconfig->return_stages();
131  auto [last_stage_name, last_stage_idx] =
132  get_last_enabled(MSTAGE, stages);
133  if (solver_stage != last_stage_idx)
134  // Deactivate current stage since we found solution
135  old_solution.set_stage(solver_stage) = false;
136  *bconfig = old_solution;
137  return exists;
138  };
139 
141  if (fs::exists(prev_abs + ".info") && fs::exists(prev_abs + ".dat")) {
142  exists = check_and_update_config(prev_abs);
143  }
144 
145  return exists;
