267  // Nucleus Radius

272

287  bconfig.set(BCO_PARAMS::MCH) = 0.5;
288 
289  bconfig.set(BCO_PARAMS::BVELX) = 0.;

291 
296  bconfig.set(BCO_PARAMS::FIXED_LAPSE) = .3;
