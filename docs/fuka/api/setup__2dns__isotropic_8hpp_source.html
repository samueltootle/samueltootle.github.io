28 void write_2dns_isotropic_init_setup_tofile(Space_polar_adapted& space,
29  config_t& bconfig,
30  tov_t& tov) {
31  using eos_t = typename tov_t::eos_t;
32 
33  enum ltpQ { LAPSE = 0, RHO, CONF };
34 
35  const int ndom = space.get_nbr_domains();
36 
37  // setup fields
38  Scalar lapse(space);
39  lapse = 1.;
40 
41  Scalar conf(lapse);
42 
43  // H = log(h), the logarithm of the specific enthalpy
44  Scalar logh(space);
45  logh.annule_hard();
46 
47  // interpolate TOV solution
48  auto lintp = setup_interpolator_from_TOV(tov);
49 
50  Scalar r_field(space);
51  r_field.annule_hard();
52  for (auto i = 0; i < ndom; ++i)
53  r_field.set_domain(i) = space.get_domain(i)->get_radius();
54  auto npts = space.get_domain(ndom - 1)->get_nbr_points();
55  Index pos_c(npts);
56  pos_c.set(0) = npts(0) - 1; // outer_bc
57 
58  for (auto i = 0; i < npts(1); ++i) {
59  pos_c.set(1) = i;
60  r_field.set_domain(ndom - 1).set(pos_c) = 1e10;
61  }
62  pos_c.set_start();
63 
64  // update the fields based on TOV solution for a given domain
65  auto update_fields = [&](const size_t dom) {
66  Index pos(space.get_domain(dom)->get_nbr_points());
67  do {
68  double rval = r_field(dom)(pos);
69  auto all_ltp = lintp.interpolate_all(rval);
70  auto rho = (all_ltp[ltpQ::RHO] <= 0) ? 1e-15 : all_ltp[ltpQ::RHO];
71  auto h = EOS<eos_t, eos_var_t::DENSITY>::h_cold__rho(rho);
72 
73  if (dom == 0 && pos(0) == 0 && pos(1) == 0)
74  bconfig.set(BCO_PARAMS::HC) = h;
75 
76  logh.set_domain(dom).set(pos) = (h <= 1) ? 0. : std::log(h);
77  lapse.set_domain(dom).set(pos) = all_ltp[ltpQ::LAPSE];
78  conf.set_domain(dom).set(pos) = all_ltp[ltpQ::CONF];
79  } while (pos.inc());
80  };
81  for (int i = 0; i < ndom; ++i)
82  update_fields(i);
83 
84  for (int d = space.ADAPTED_INNER; d < ndom; ++d)
85  logh.set_domain(d).annule_hard();
86 
87  // Fix compactified domain and additional shells
88  for (auto d = 3; d < ndom; ++d) {
89  pos_c.set_start();
90  auto decay_factor = r_field(d)(pos_c) / r_field(d);
91  if (d > 3) {
92  pos_c.set(0) = npts(0) - 1;
93  conf.set_domain(d) = 1 + (conf(d - 1)(pos_c) - 1) * decay_factor;
94  lapse.set_domain(d) = 1 + (lapse(d - 1)(pos_c) - 1) * decay_factor;
95  } else {
96  conf.set_domain(d) = 1 + (conf(d)(pos_c) - 1) * decay_factor;
97  lapse.set_domain(d) = 1 + (lapse(d)(pos_c) - 1) * decay_factor;
98  }
99  }
100 
101  Scalar A(conf * conf);
102  A.std_base();
103  lapse.std_base();
104 
105  // We store the quantities that appear in the lapace terms
106  // for solver stability. These will need to be transformed
107  // to obtain the full metric.
108  Scalar nu(log(lapse));
109  Scalar lap_aterm(nu + log(A));
110 
111  logh.std_base();
112  nu.std_base();
113  lap_aterm.std_base();
114 
115  Scalar tmp(lapse * A - 1);
116  Scalar lap_Bterm = Scalar(tmp.mult_r().mult_sin_theta());
117  // end setup fields
118 
119  bco_utils::save_to_file(space, bconfig, lap_aterm, nu, logh, lap_Bterm);
120 }
