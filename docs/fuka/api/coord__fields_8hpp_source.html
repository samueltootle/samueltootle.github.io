298 template <typename space_t>
299 template <int ind_t>
300 Kadath::Vector CoordFields<space_t>::cart(double shift_x,
301  double shift_y,
302  double shift_z) const {
303  Kadath::Vector cart(space, ind_t, basis);
304 
305  int ndom = space.get_nbr_domains();
306 
307  for (int d = 0; d < ndom - 1; d++) {
308  cart.set(1).set_domain(d) = space.get_domain(d)->get_cart(1) - shift_x;
309  cart.set(2).set_domain(d) = space.get_domain(d)->get_cart(2) - shift_y;
310  cart.set(3).set_domain(d) = space.get_domain(d)->get_cart(3) - shift_z;
311  }
312 
313  // set outter boundary to surface coordinates (1/r) for compact domain
314  // FIXME: these are not coordinate shifted
315  cart.set(1).set_domain(ndom - 1) =
316  space.get_domain(ndom - 1)->get_cart_surr(1);
317  cart.set(2).set_domain(ndom - 1) =
318  space.get_domain(ndom - 1)->get_cart_surr(2);
319  cart.set(3).set_domain(ndom - 1) =
320  space.get_domain(ndom - 1)->get_cart_surr(3);
321 
322  cart.set(1).std_base();
323  cart.set(2).std_base();
324  cart.set(3).std_anti_base();
325 
326  return cart;
327 }

329 template <typename space_t>
330 Kadath::Scalar CoordFields<space_t>::radius(double shift_x,
331  double shift_y,
332  double shift_z) const {
333  auto coords = this->cart(shift_x, shift_y, shift_z);
334  int ndom = space.get_nbr_domains();
335 
336  Kadath::Scalar r_sq = coords(1) * coords(1) + coords(2) * coords(2) +
337  coords(3) * coords(3);
338  r_sq.std_base();
339 
340  Kadath::Scalar r = sqrt(r_sq);
341  r.std_base();
342 
343  // set outter boundary to constant large radius
344  Index pos(space.get_domain(ndom - 1)->get_nbr_points());
345  const int npts_r = space.get_domain(ndom - 1)->get_nbr_points()(0);
346  for (int i = 1; i <= 3; ++i) {
347  const int dom = ndom - 1;
348  do {
349  if (pos(0) == npts_r - 1) {
350  r.set_domain(dom).set(pos) = 1e10;
351  }
352  } while (pos.inc());
353  }
354 
355  return r;
356 }

374 template <typename space_t>
375 template <int ind_t>
376 Kadath::Vector CoordFields<space_t>::e_rad(double shift_x,
377  double shift_y,
378  double shift_z) const {
379  auto coords = this->cart(shift_x, shift_y, shift_z);
380 
381  int ndom = space.get_nbr_domains();
382 
383  // this can be problematic around the given origin
384  Kadath::Vector e_rad(space, ind_t, basis);
385  for (int i : {1, 2, 3}) {
386  e_rad.set(i) = coords(i) / this->radius(shift_x, shift_y, shift_z);
387 
388  // fix non-finite values likely at (0,0,0)
389  // Note: this fix is sufficient for the FUKA codes as
390  // we are only concerned with e_rad on the surfaces of compact
391  // objects or at infinity. This fix is to remove NaNs and their
392  // impact on the system of equations
393  for (int dom = 0; dom < ndom - 1; ++dom) {
394  Index pos(space.get_domain(dom)->get_nbr_points());
395  __attribute__((unused))
396  const int npts_r = space.get_domain(dom)->get_nbr_points()(0);
397  do {
398  if (!std::isfinite(e_rad(i)(dom)(pos)) && pos(0) == 0) {
399  Index tempos(pos);
400  tempos.set(0) = pos(0) + 1;
401  e_rad.set(i).set_domain(dom).set(pos) =
402  e_rad(i)(dom)(tempos);
403  }
404  } while (pos.inc());
405  }
406 
407  // FIXME this is not correct for shifted cartesian coords
408  e_rad.set(i).set_domain(ndom - 1) =
409  space.get_domain(ndom - 1)->get_cart_surr(i);
410  }
411 
412  e_rad.set(1).std_base();
413  e_rad.set(2).std_base();
414  e_rad.set(3).std_anti_base();
415 
416  return e_rad;
417 }

435 template <typename space_t>
436 void update_fields(CoordFields<space_t> const& cf_generator,
437  vec_ary_t& coord_vectors,
438  scalar_ary_t& coord_scalars,
439  const double xo,
440  const double xc1,
441  const double xc2,
442  std::unique_ptr<System_of_eqs>& syst) {
443 
444  if (coord_vectors[GLOBAL_ROT])
445  *coord_vectors[GLOBAL_ROT] = cf_generator.template rot_z<>(xo);
446  if (coord_vectors[BCO1_ROT])
447  *coord_vectors[BCO1_ROT] = cf_generator.template rot_z<>(xc1);
448  if (coord_vectors[BCO2_ROT])
449  *coord_vectors[BCO2_ROT] = cf_generator.template rot_z<>(xc2);
450  if (coord_vectors[EX])
451  *coord_vectors[EX] = cf_generator.template e_cart<COV>(1);
452  if (coord_vectors[EY])
453  *coord_vectors[EY] = cf_generator.template e_cart<COV>(2);
454  if (coord_vectors[EZ])
455  *coord_vectors[EZ] = cf_generator.template e_cart<COV>(3);
456  if (coord_vectors[S_BCO1])
457  *coord_vectors[S_BCO1] = cf_generator.template e_rad<COV>(xc1);
458  if (coord_vectors[S_BCO2])
459  *coord_vectors[S_BCO2] = cf_generator.template e_rad<COV>(xc2);
460  if (coord_vectors[S_INF])
461  *coord_vectors[S_INF] = cf_generator.template e_rad<COV>(xo);
462  if (coord_scalars[R_BCO1])
463  *coord_scalars[R_BCO1] = cf_generator.radius(xc1);
464  if (coord_scalars[R_BCO2])
465  *coord_scalars[R_BCO2] = cf_generator.radius(xc2);
466 
467  int ndom = coord_vectors[GLOBAL_ROT]->get_space().get_nbr_domains();
468 
469  auto update = [&](auto& name, auto& field) {
470  for (int dom = 0; dom < ndom; ++dom) {
471  __attribute__((unused)) bool succ =
472  update_field(*syst, dom, name.c_str(), field);
473 #ifdef DEBUG
474  if (!succ)
475  std::cout << name << " failed\n";
476 #endif
477  }
478  };
479  if (syst) {
480  for (int i = 0; i < NUM_VECTORS; ++i) {
481  if (coord_vectors[i])
482  update(cv_names[i], *coord_vectors[i]);
483  }
484  for (int i = 0; i < NUM_SCALARS; ++i) {
485  if (coord_scalars[i])
486  update(cs_names[i], *coord_scalars[i]);
487  }
488  syst->sec_member();
489  }
490 }

582 inline bool update_field(System_of_eqs& syst,
583  int dom,
584  const char* n,
585  Tensor& so) {
586  char* name;
587  name = new char[LMAX];
588  trim_spaces(name, n);
589  int which = -1;
590  int valence;
591  char* name_ind = 0x0;
592  Array<int>* type_ind = 0x0;
593  bool found = false;
594 
595  found = syst.iscst(name, which, valence, name_ind, type_ind);
596  if (found) {
597  Term_eq* mm = syst.give_cst(which, dom);
598  mm->set_val_t(so);
599  }
600  return found;
601 }

603 template <class space_t>
604 vec_ary_t default_binary_vector_ary(space_t& space) {
605  vec_ary_t coord_vectors{};
