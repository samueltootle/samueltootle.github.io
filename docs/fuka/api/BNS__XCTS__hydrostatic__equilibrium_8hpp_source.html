36 template <class eos_t>
37 void BNS_XCTS<eos_t>::setup_hydrostatic_equilibrium_stage() {
38  if (rank == 0) {
39  std::cout << std::string(42, '#') << std::endl
40  << "TOTAL - Hydrostatic equilibrium stage\n"
41  << "with mass fixing, chi = "
42  << (*bconfig)(BCO_PARAMS::CHI, NODES::BCO1) << ","
43  << (*bconfig)(BCO_PARAMS::CHI, NODES::BCO2)
44  << " and q = " << (*bconfig)(Q) << std::endl
45  << std::string(42, '#') << std::endl;
46  }
47  // We use `config_filename()` vs `config_filename_abs()` since
48  // `solution_exists` will probe the HOME_KADATH/COs directory
49  auto const current = bconfig->config_filename();
50  if (!bconfig->control(RESOLVE) && solution_exists()) {
51  if (rank == 0) {
52  std::cout << "Solved previously: " << bconfig->config_filename_abs()
53  << std::endl;
54  }
55  reload();
56  }
57 
58  {
59  double loghc = bco_u::get_boundary_val(space->NS1, *logh, INNER_BC);
60  bconfig->set(BCO_PARAMS::HC, NODES::BCO1) = std::exp(loghc);
61  }
62  {
63  double loghc = bco_u::get_boundary_val(space->NS2, *logh, INNER_BC);
64  bconfig->set(BCO_PARAMS::HC, NODES::BCO2) = std::exp(loghc);
65  }
66 
67  // setup background position vector field - only needed for ECC_RED stage
68  Kadath::update_fields(*cfields, *coord_vectors, {}, xo, xc1, xc2);
69 
70  // Ensure logh_const is free
71  logh_const.reset(nullptr);
72 
73  // setup a system of equations
74  syst.reset(new System_of_eqs(*space, 0, ndom - 1));
75 
76  // the actual solution fields of the equations, i.e.
77  // conformal factor, lapse, shift and (logarithmic) enthalpy
78  // set this first before running syst_init()
79  syst->add_var("H", *logh);
80 
81  // central enthalpy is a variable, fixed by the baryonic mass
82  // integral
83  syst->add_var("hc1", (*bconfig)(BCO_PARAMS::HC, NODES::BCO1));
84  syst->add_var("hc2", (*bconfig)(BCO_PARAMS::HC, NODES::BCO2));
85 
86  Param p;
87  Kadath::FUKA_EOS::set_eos_ope_struct<eos_t>()(*syst, p);
88 
89  // populate all the boiler-plate constants, variables, and definitions
90  syst_init();
91 
92  // center of mass on the x-axis connecting both companions
93  // and orbital angular frequency parameter
94  // in this case, both are fixed by the two central force-balance
95  // equations
96  syst->add_var("ome", (*bconfig)(BIN_PARAMS::GOMEGA));
97  syst->add_var("xaxis", (*bconfig)(BIN_PARAMS::COM));
98 
99  // orbital rotation vector field, corrected by the "center of mass" shift
100  syst->add_def("Morb^i = mg^i + xaxis * ey^i");
101 
102  // full shift vector field, incoporating the orbital part
103  syst->add_def("B^i= bet^i + ome * Morb^i");
104 
105  // the actual equations, defined differently in the different domains
106  for (int d = 0; d < ndom; d++) {
107  // if outside the stellar domains, without matter sources
108  // resort to the source-free constraint equations
109  // and set matter (and velocity potential) to zero
110  if ((d >= space->ADAPTED2 + 1) ||
111  ((d >= space->ADAPTED1 + 1) && (d < space->NS2))) {
112  if (!bconfig->control(CONTROLS::COROT_BIN))
113  syst->add_eq_full(d, "phi= 0");
114 
115  syst->add_eq_full(d, "H = 0");
116  syst->add_def(d, "eqP = D^i D_i P + A_ij * A^ij / P^7 / 8");
117  syst->add_def(
118  d,
119  "eqNP = D^i D_i NP - 7. / 8. * NP / P^8 * A_ij * A^ij");
120  syst->add_def(
121  d,
122  "eqbet^i = D_j D^j bet^i + D^i D_j bet^j / 3. - 2. * A^ij * "
123  "D_j Ntilde");
124 
125  }
126  // in case of the domains harboring the two stars
127  // define all derived matter related quantities
128  // and enforce the transformed contraint equations (i.e. multiplied by p /
129  // rho)
130  else {
131  // 3-velocity and first integral of the Euler equation
132  // in case of corotation
133  if (bconfig->control(CONTROLS::COROT_BIN)) {
134  syst->add_def(d, "U^i = B^i / N");
135  syst->add_def(d, "Usquare= P^4 * U_i * U^i");
136  syst->add_def(d, "Wsquare= 1 / (1 - Usquare)");
137  syst->add_def(d, "W = sqrt(Wsquare)");
138  syst->add_def(d, "firstint = log(h * N / W)");
139  }
140  // 3-velocity, (approximate) first integral of the Euler equation
141  // and velocity potential equation
142  // in case of irrotational or spinning companions
143  else {
144  syst->add_def(d, "Wsquare= eta^i * eta_i / h^2 / P^4 + 1.");
145  syst->add_def(d, "W = sqrt(Wsquare)");
146  syst->add_def(d, "U^i = eta^i / P^4 / h / W");
147  syst->add_def(d, "Usquare= P^4 * U_i * U^i");
148  syst->add_def(d, "V^i = N * U^i - B^i");
149  syst->add_def(d, "firstint = log(h * N / W + D_i phi * V^i)");
150  syst->add_def(d,
151  "eqphi = P^6 * W * V^i * D_i H + dHdlnrho * D_i "
152  "(P^6 * W * V^i)");
153  }
154  // transformed source terms
155  syst->add_def(d, "Etilde = press * h * Wsquare - press * delta");
156  syst->add_def(d,
157  "Stilde = 3 * press * delta + (Etilde + press * "
158  "delta) * Usquare");
159  syst->add_def(d, "ptilde^i = press * h * Wsquare * U^i");
160 
161  // transformed constraint equations
162  syst->add_def(
163  d,
164  "eqP = delta * D^i D_i P + A_ij * A^ij / P^7 / 8 * delta "
165  "+ 4piG / 2. * P^5 * Etilde");
166  syst->add_def(
167  d,
168  "eqNP = delta * D^i D_i NP - 7. / 8. * NP / P^8 * delta * "
169  "A_ij *A^ij "
170  "- 4piG / 2. * N * P^5 * (Etilde + 2. * Stilde)");
171  syst->add_def(
172  d,
173  "eqbet^i= delta * D_j D^j bet^i + delta * D^i D_j bet^j / 3. "
174  "- 2. * delta * A^ij * D_j Ntilde - 4. * 4piG * N * P^4 * "
175  "ptilde^i");
176 
177  // baryonic mass volume integrant
178  syst->add_def(d, "intMb = P^6 * rho * W");
179  // quasi-local ADM mass volume integrant
180  syst->add_def(d, "intM = - D_i D^i P * 2. / 4piG");
181  }
182  }
183  // add equations to the system and demand continuity
184  // along the normal of the domains
185  space->add_eq(*syst, "eqNP= 0", "N", "dn(N)");
186  space->add_eq(*syst, "eqP= 0", "P", "dn(P)");
187  space->add_eq(*syst, "eqbet^i= 0", "bet^i", "dn(bet^i)");
188 
189  // boundary conditions at infinity
190  syst->add_eq_bc(ndom - 1, OUTER_BC, "N=1");
191  syst->add_eq_bc(ndom - 1, OUTER_BC, "P=1");
192  syst->add_eq_bc(ndom - 1, OUTER_BC, "bet^i=0");
193 
194  // boundary conditions defining the boundary of the adapted domains, i.e.
195  // vanishing matter
196  syst->add_eq_bc(space->ADAPTED1, OUTER_BC, "H = 0");
197  syst->add_eq_bc(space->ADAPTED2, OUTER_BC, "H = 0");
198 
199  // determine equations to add based on corotation
200  // or mixed spin binaries based on chi or fixed omega
201  if (!bconfig->control(CONTROLS::COROT_BIN)) {
202  syst->add_eq_bc(space->ADAPTED1, OUTER_BC, "V^i * D_i H = 0");
203  syst->add_eq_bc(space->ADAPTED2, OUTER_BC, "V^i * D_i H = 0");
204 
205  for (int i = space->NS1; i < space->ADAPTED1; ++i) {
206  syst->add_eq_vel_pot(i, 2, "eqphi = 0", "phi=0");
207  syst->add_eq_matching(i, OUTER_BC, "phi");
208  syst->add_eq_matching(i, OUTER_BC, "dn(phi)");
209  }
210  syst->add_eq_vel_pot(space->ADAPTED1, 2, "eqphi = 0", "phi=0");
211 
212  for (int i = space->NS2; i < space->ADAPTED2; ++i) {
213  syst->add_eq_vel_pot(i, 2, "eqphi = 0", "phi=0");
214  syst->add_eq_matching(i, OUTER_BC, "phi");
215  syst->add_eq_matching(i, OUTER_BC, "dn(phi)");
216  }
217  syst->add_eq_vel_pot(space->ADAPTED2, 2, "eqphi = 0", "phi=0");
218 
219  if (std::isnan(bconfig->set(BCO_PARAMS::FIXED_BCOMEGA, NODES::BCO1)))
220  space->add_eq_int_outer_sphere_one(
221  *syst,
222  "integ(intS1) / Madm1 / Madm1 = chi1");
223 
224  if (std::isnan(bconfig->set(BCO_PARAMS::FIXED_BCOMEGA, NODES::BCO2)))
225  space->add_eq_int_outer_sphere_two(
226  *syst,
227  "integ(intS2) / Madm2 / Madm2 = chi2");
228  }
229 
230  // force-balance equations at the center of each star
231  // to fix the orbital frequency as well as the "center of mass"
232  // (the latter can get very inaccurate in extreme configurations
233  // with large residual linear momenta, for which the next stage is given)
234  Index posori1(space->get_domain(space->NS1)->get_nbr_points());
235  syst->add_eq_val(space->NS1, "ex^i * D_i H", posori1);
236 
237  Index posori2(space->get_domain(space->NS2)->get_nbr_points());
238  syst->add_eq_val(space->NS2, "ex^i * D_i H", posori2);
239 
240  // add the first integral to get (approximate) hydrostatic equilibrium
241  syst->add_eq_first_integral(space->NS1,
242  space->ADAPTED1,
243  "firstint",
244  "h - hc1");
245  syst->add_eq_first_integral(space->NS2,
246  space->ADAPTED2,
247  "firstint",
248  "h - hc2");
249 
250  space->add_eq_int_volume(*syst,
251  space->NS1,
252  space->ADAPTED1,
253  "integvolume(intMb) = Mb1");
254 
255  space->add_eq_int_volume(*syst,
256  space->NS2,
257  space->ADAPTED2,
258  "integvolume(intMb) = Mb2");
259 }
